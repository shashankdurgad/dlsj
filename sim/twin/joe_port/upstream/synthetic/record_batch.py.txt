RAW=SYN/'raw';RAW.mkdir(exist_ok=True);started=time.monotonic();report=[]
for path in sorted(RAW.glob('candidate-*/result.json'),key=lambda p:int(p.parent.name.split('-')[1])):report.append(json.loads(path.read_text()))
counts={s:sum(r['success'] and r['split']==s for r in report) for s in ['train','validation']}
completed={r['candidate'] for r in report}
rep.orchestrator.set_capture_on_play(False)
render_products['overview'].hydra_texture.set_updates_enabled(False)
# Warm up Replicator before resetting the first episode.
await rep.orchestrator.step_async(rt_subframes=1,pause_timeline=False,delta_time=0.,wait_for_render=True)
distractors={}
for name,yaw in [('blue_brick',120),('yellow_brick',77),('cream_brick',50)]:
 distractors[name]=(RigidPrim('/World/ReferenceProps/'+name),reference['objects'][name]['position'],np.deg2rad(yaw))

def canonical(q):
 state=np.rad2deg(np.asarray(q)-joint_offsets);state[-1]=q[-1]/limits[indices[-1],1]*100;return state.astype(np.float32)

def pose_reset(view,position,yaw):
 view.set_world_poses(positions=[position],orientations=[[np.cos(yaw/2),0,0,np.sin(yaw/2)]])
 view.set_velocities(linear_velocities=[[0,0,0]],angular_velocities=[[0,0,0]])

def encoder(path):
 return subprocess.Popen([str(Path.home()/'.local/bin/ffmpeg'),'-hide_banner','-loglevel','error','-y','-f','rawvideo','-pixel_format','rgb24','-video_size','640x480','-framerate','30','-i','-','-an','-c:v','libx264','-threads','1','-g','2','-pix_fmt','yuv420p','-crf','18',str(path)],stdin=subprocess.PIPE)

def progress(status,candidate=None,frame=None):
 (SYN/'generation-progress.json').write_text(json.dumps({'status':status,'counts':counts,'targets':config,'attempts':len(report),'current_candidate':candidate,'frame':frame,'elapsed_s':time.monotonic()-started},indent=2))

try:
 for plan in plans:
  candidate=plan['candidate'];split=plan['split']
  if candidate in completed or counts[split]>=config[split]:continue
  directory=RAW/f'candidate-{candidate:03d}';directory.mkdir(exist_ok=True)
  timeline.play();q0=np.asarray(plan['initial_q'],np.float32)
  robot.set_dof_positions(q0[None],dof_indices=indices);robot.set_dof_velocities(np.zeros((1,6)),dof_indices=indices);robot.set_dof_position_targets(q0[None],dof_indices=indices)
  pose_reset(brick_view,plan['target_position'],np.deg2rad(plan['target_yaw_deg']))
  for view,position,yaw in distractors.values():pose_reset(view,position,yaw)
  light.GetIntensityAttr().Set(700*plan['light_scale'])

  initial=brick_view.get_world_poses()[0].numpy()[0].copy();dist_initial={k:v[0].get_world_poses()[0].numpy()[0].copy() for k,v in distractors.items()};max_dist_disp={k:0. for k in distractors};positions=[];observations=[];actions=[];raw_q=[];arm_errors=[];camera_std={c:[] for c in ['wrist','top','inspect']};extra_steps=0
  encoders={c:encoder(directory/f'{c}.mp4') for c in ['wrist','top','inspect']}
  first_step=SimulationManager.get_num_physics_steps()
  try:
   for i,(action,phase) in enumerate(zip(plan['actions'],plan['phases'])):

    extra_steps+=SimulationManager.get_num_physics_steps()-before

    q=robot.get_dof_positions().numpy()[0,indices].copy()

    for camera,e in encoders.items():
     pixels=np.asarray(annotators[camera].get_data())[:,:,:3].copy();assert pixels.shape==(480,640,3)

  (directory/'result.json').write_text(json.dumps(result,indent=2));report.append(result)
