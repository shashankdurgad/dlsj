def linear_color(rgb):
    a=np.asarray(rgb);return np.where(a<=.04045,a/12.92,((a+.055)/1.055)**2.4).tolist()

def material(name,rgb,roughness=.5,texture=None,texture_scale=(1,1)):
    path='/World/ReferenceMaterials/'+name
    m=UsdShade.Material.Define(stage,path);shader=UsdShade.Shader.Define(stage,path+'/Surface');shader.CreateIdAttr('UsdPreviewSurface')
    shader.CreateInput('diffuseColor',Sdf.ValueTypeNames.Color3f).Set(Gf.Vec3f(*linear_color(rgb)))
    shader.CreateInput('roughness',Sdf.ValueTypeNames.Float).Set(roughness)
    if texture:
        reader=UsdShade.Shader.Define(stage,path+'/UV');reader.CreateIdAttr('UsdPrimvarReader_float2');reader.CreateInput('varname',Sdf.ValueTypeNames.Token).Set('st')
        transform=UsdShade.Shader.Define(stage,path+'/Transform');transform.CreateIdAttr('UsdTransform2d');transform.CreateInput('in',Sdf.ValueTypeNames.Float2).ConnectToSource(reader.ConnectableAPI(),'result');transform.CreateInput('scale',Sdf.ValueTypeNames.Float2).Set(Gf.Vec2f(*texture_scale))
        tex=UsdShade.Shader.Define(stage,path+'/Texture');tex.CreateIdAttr('UsdUVTexture');tex.CreateInput('file',Sdf.ValueTypeNames.Asset).Set(Sdf.AssetPath(str(texture)));tex.CreateInput('sourceColorSpace',Sdf.ValueTypeNames.Token).Set('sRGB');tex.CreateInput('st',Sdf.ValueTypeNames.Float2).ConnectToSource(transform.ConnectableAPI(),'result');tex.CreateInput('wrapS',Sdf.ValueTypeNames.Token).Set('mirror');tex.CreateInput('wrapT',Sdf.ValueTypeNames.Token).Set('mirror');shader.GetInput('diffuseColor').ConnectToSource(tex.ConnectableAPI(),'rgb')
    m.CreateSurfaceOutput().ConnectToSource(shader.ConnectableAPI(),'surface');return m

def cube(path,position,scale,mat,collision=True,yaw=0):
    c=UsdGeom.Cube.Define(stage,path);c.CreateSizeAttr(1)
    c.AddTranslateOp().Set(Gf.Vec3d(*position))
    if yaw:c.AddRotateZOp().Set(yaw)
    c.AddScaleOp().Set(Gf.Vec3f(*scale));bind(c.GetPrim(),mat)
    if collision:UsdPhysics.CollisionAPI.Apply(c.GetPrim())
    return c

def cylinder(path,pos,radius,height,mat,collision=False):
    c=UsdGeom.Cylinder.Define(stage,path);c.CreateRadiusAttr(radius);c.CreateHeightAttr(height);c.CreateAxisAttr('Z');c.AddTranslateOp().Set(Gf.Vec3d(*pos));bind(c.GetPrim(),mat)
    if collision:UsdPhysics.CollisionAPI.Apply(c.GetPrim())
    return c

def ring_mesh(path,rings,mat,segments=64):
    verts=[(radius*math.cos(2*math.pi*i/segments),radius*math.sin(2*math.pi*i/segments),z) for radius,z in rings for i in range(segments)]
    faces=[]
    for j in range(len(rings)-1):
        for i in range(segments):faces.extend([j*segments+i,j*segments+(i+1)%segments,(j+1)*segments+(i+1)%segments,(j+1)*segments+i])
    m=UsdGeom.Mesh.Define(stage,path);m.CreatePointsAttr(verts);m.CreateFaceVertexCountsAttr([4]*(len(faces)//4));m.CreateFaceVertexIndicesAttr(faces);m.CreateSubdivisionSchemeAttr('none');bind(m.GetPrim(),mat);return m

def plate(name,obj,mat):
    root='/World/ReferenceProps/'+name

    x=UsdGeom.Xform.Define(stage,root);x.AddTranslateOp().Set(Gf.Vec3d(*p))

        cube(root+f'/Rim{i}',(.087*math.cos(theta),.087*math.sin(theta),.009),(.012,.025,.018),mat,True,theta*180/math.pi)

def brick(name,obj,yaw,studs=True):
    root='/World/Brick' if name=='target' else '/World/ReferenceProps/'+name
    mat=material('Brick_'+name,obj['color'],.35)
    parent=UsdGeom.Xform.Define(stage,root);parent.AddTranslateOp().Set(Gf.Vec3d(*obj['position']));parent.AddRotateZOp().Set(yaw)
    UsdPhysics.RigidBodyAPI.Apply(parent.GetPrim());UsdPhysics.MassAPI.Apply(parent.GetPrim()).CreateMassAttr(.012)
    size=obj['size'];cube(root+'/Body',(0,0,0),size,mat)
    if studs:
        count_x=int(round(size[0]/.016));count_y=int(round(size[1]/.016))
        for ix in range(count_x):
            for iy in range(count_y):
                stud=ring_mesh(root+f'/Stud{ix}_{iy}',[(.0047,0),(.0047,.0044),(.0033,.0044),(.0033,.0001)],mat,32)
                stud.AddTranslateOp().Set(Gf.Vec3d((ix-(count_x-1)/2)*.016,(iy-(count_y-1)/2)*.016,size[2]/2))
    return parent

def set_camera(path,params,local=False):
    cam=UsdGeom.Camera(stage.GetPrimAtPath(path));cam.CreateFocalLengthAttr(20.)
    f=params['focal_px'];cam.CreateHorizontalApertureAttr(20.*640/f);cam.CreateVerticalApertureAttr(20.*480/f);cam.CreateClippingRangeAttr(Gf.Vec2f(.002,30))

    xf=UsdGeom.Xformable(cam);xf.ClearXformOpOrder();xf.AddTransformOp().Set(Gf.Matrix4d(matrix.tolist()))
