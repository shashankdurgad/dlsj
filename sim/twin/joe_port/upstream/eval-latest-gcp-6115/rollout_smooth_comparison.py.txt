def smooth_chunk_arm(chunk_arm: np.ndarray) -> np.ndarray:
    kernel = np.array([1.0, 4.0, 6.0, 4.0, 1.0], dtype=np.float64) / 16.0
    pad = np.pad(chunk_arm, ((2, 2), (0, 0)), mode='edge')
    out = np.zeros_like(chunk_arm, dtype=np.float64)
    for k_i, w in enumerate(kernel):
        out += w * pad[k_i : k_i + len(chunk_arm)]
    return out

class SmoothController:
    def __init__(self, mode='passthrough', exec_horizon=50, dt=1.0 / 30.0, blend_tau_steps=3.0):
        self.mode = mode
        self.exec_horizon = int(exec_horizon)
        self.dt = float(dt)
        self.du_max = V_MAX_ARM_DEG_S * self.dt
        self.ddu_max = A_MAX_ARM_DEG_S2 * (self.dt ** 2)
        self.blend_tau = float(blend_tau_steps)
        self.u_prev = None
        self.du_prev = np.zeros(5, dtype=np.float64)
        self.raw_chunk = None
        self.chunk_ref_arm = None

    def reset(self, initial_state_canonical: np.ndarray):
        self.u_prev = np.asarray(initial_state_canonical[:5], dtype=np.float64).copy()
        self.du_prev = np.zeros(5, dtype=np.float64)
        self.raw_chunk = None
        self.chunk_ref_arm = None

    def start_chunk(self, raw_chunk_actions: np.ndarray, current_state_canonical: np.ndarray):
        raw = np.asarray(raw_chunk_actions, dtype=np.float64)
        self.raw_chunk = raw.copy()
        if self.u_prev is None:
            self.reset(current_state_canonical)
        if self.mode == 'passthrough':
            self.chunk_ref_arm = raw[:, :5].copy()
        else:
            arm_base = smooth_chunk_arm(raw[:, :5])
            offset = self.u_prev - arm_base[0]
            steps = np.arange(1, len(arm_base) + 1, dtype=np.float64)[:, None]
            decay = np.exp(-steps / self.blend_tau)
            self.chunk_ref_arm = arm_base + offset[None, :] * decay

    def step(self, step_in_chunk: int):
        raw_act = self.raw_chunk[step_in_chunk].astype(np.float32).copy()
        if self.mode == 'passthrough':
            self.u_prev = self.raw_chunk[step_in_chunk, :5].copy()
            return raw_act, raw_act.copy()
        ref_arm = self.chunk_ref_arm[step_in_chunk]
        du_des = ref_arm - self.u_prev
        du_acc = np.clip(du_des, self.du_prev - self.ddu_max, self.du_prev + self.ddu_max)
        du = np.clip(du_acc, -self.du_max, self.du_max)
        u_next = self.u_prev + du
        self.du_prev = du
        self.u_prev = u_next
        grip_cmd = float(self.raw_chunk[step_in_chunk, 5])
        applied_act = np.r_[u_next, grip_cmd].astype(np.float32)
        return raw_act, applied_act

def canonical_state(offsets_rad):
    values = robot.get_dof_positions().numpy()[0, indices].copy()
    state = np.rad2deg(values - offsets_rad)
    state[-1] = values[-1] / limits[indices[-1], 1] * 100.0
    return state.astype(np.float32), values

def action_to_sim_radians(action, offsets_rad):
    target = np.deg2rad(action).astype(np.float32) + offsets_rad
    target[-1] = np.clip(action[-1] / 100.0, 0.0, 1.0) * limits[indices[-1], 1]
    return np.clip(target, limits[indices, 0], limits[indices, 1]).astype(np.float32)

def pose_reset(view, position, yaw):
    view.set_world_poses(
        positions=[np.asarray(position, np.float32)],
        orientations=[[np.cos(yaw / 2.0), 0.0, 0.0, np.sin(yaw / 2.0)]],
    )
    view.set_velocities(linear_velocities=[[0.0, 0.0, 0.0]], angular_velocities=[[0.0, 0.0, 0.0]])

def open_video_encoder(path, width=640, height=480, crf=18):

        [
            ffmpeg_bin, '-hide_banner', '-loglevel', 'error', '-y',
            '-f', 'rawvideo', '-pixel_format', 'rgb24',
            '-video_size', f'{width}x{height}', '-framerate', '30',
            '-i', '-', '-an', '-c:v', 'libx264', '-threads', '1',
            '-g', '2', '-pix_fmt', 'yuv420p', '-crf', str(crf), str(path),
        ],
        stdin=subprocess.PIPE,
    )
