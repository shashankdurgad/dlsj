    def joint_state():
        values=robot.get_dof_positions().numpy()[0,indices].copy()
        state=np.rad2deg(values)
        state[-1]=values[-1]/limits[indices[-1],1]*100
        return state.astype(np.float32)
    
    records=[]
    observed=[]
    requested=[]
    applied=[]
    frame_index=0

    for chunk in range(10):
        # Freeze simulation during synchronous inference, so observations cannot become stale.
        timeline.pause()
        await app_utils.update_app_async(steps=2)
        observation={'state':joint_state(),'instruction':np.asarray(instruction)}
        for name in ['wrist','top','inspect']:
            pixels=np.asarray(annotators[name].get_data())[:,:,:3].copy()
            assert pixels.shape==(480,640,3),(name,pixels.shape)
            observation[name]=pixels
        temp=IPC/f'obs-{chunk:03d}.tmp.npz'
        np.savez(temp,**observation)
        os.replace(temp,IPC/f'obs-{chunk:03d}.npz')
        actionfile=IPC/f'actions-{chunk:03d}.npz'
        deadline=time.monotonic()+600
        while not actionfile.exists():
            assert time.monotonic()<deadline, 'Policy worker timed out'
            await asyncio.sleep(.05)
        result=np.load(actionfile,allow_pickle=False)
        actions=result['actions']
        assert actions.shape==(50,6) and np.isfinite(actions).all()
        records.append({'chunk':chunk,'inference_s':float(result['latency_s']),'state':observation['state'].tolist()})
        timeline.play()
        for action in actions:
            target=np.deg2rad(action).astype(np.float32)
            target[-1]=np.clip(action[-1]/100,0,1)*limits[indices[-1],1]
            target=np.clip(target,limits[indices,0],limits[indices,1])
            robot.set_dof_position_targets(target.reshape(1,-1),dof_indices=indices)
            await app_utils.update_app_async(steps=2)
            pixels=np.asarray(annotators['overview'].get_data())[:,:,:3]
            Image.fromarray(pixels).save(frames/f'{frame_index:05d}.png')
            observed.append(joint_state())
            requested.append(action.copy())
            applied.append(target.copy())
            frame_index+=1
        print('CHUNK_EXECUTED',chunk,flush=True)
    timeline.pause()

        Image.fromarray(np.asarray(ann.get_data())[:,:,:3]).save(OUT/f'final-{name}.png')
