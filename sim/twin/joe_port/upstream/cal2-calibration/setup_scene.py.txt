"""Run through Isaac Sim's localhost Python server, in a named context."""
import asyncio
import json
from pathlib import Path
import numpy as np
import omni.kit.app
import omni.usd
import omni.timeline
import omni.replicator.core as rep
from pxr import Usd, UsdGeom, UsdLux, UsdPhysics, Gf
import isaacsim.core.experimental.utils.stage as stage_utils
import isaacsim.core.experimental.utils.app as app_utils
from isaacsim.core.simulation_manager import SimulationManager

ROOT = Path.home() / 'pi05-isaac-so101'
OUT = ROOT / 'cal2-calibration' / 'setup'
OUT.mkdir(parents=True, exist_ok=True)
JOINTS = ['shoulder_pan','shoulder_lift','elbow_flex','wrist_flex','wrist_roll','gripper']
await stage_utils.create_new_stage_async(template='empty')
manager = omni.kit.app.get_app().get_extension_manager()
for ext in ['omni.scene.optimizer.core','isaacsim.robot.schema','isaacsim.asset.importer.urdf']:
    manager.set_extension_enabled_immediate(ext, True)
from isaacsim.asset.importer.urdf.impl import URDFImporter, URDFImporterConfig
asset_dir = ROOT / 'assets' / 'converted'
asset_dir.mkdir(parents=True, exist_ok=True)
config = URDFImporterConfig(
    urdf_path=str(ROOT/'robot-source/Simulation/SO101/so101_new_calib.urdf'),
    usd_path=str(asset_dir), fix_base=True, merge_fixed_joints=True,
    collision_from_visuals=True, joint_drive_type='force', joint_target_type='position',
    override_joint_stiffness=50.0, override_joint_damping=3.0,
    robot_type='Manipulator', run_asset_transformer=False,
    run_multi_physics_conversion=True,
)
usd_path = URDFImporter(config).import_urdf()
print('IMPORTED', str(usd_path), flush=True)
stage = omni.usd.get_context().get_stage()
robot_prim = stage_utils.add_reference_to_stage(usd_path=str(usd_path),path='/World/SO101')
UsdGeom.SetStageMetersPerUnit(stage, 1.0)
UsdGeom.SetStageUpAxis(stage, UsdGeom.Tokens.z)
light = UsdLux.DomeLight.Define(stage,'/World/Light')
light.CreateIntensityAttr(1500)
sun=UsdLux.DistantLight.Define(stage,'/World/Sun')
sun.CreateIntensityAttr(600)
UsdGeom.Xformable(sun).AddRotateXYZOp().Set(Gf.Vec3f(25,-30,20))

def box(path, pos, scale, color, dynamic=False):
    geom=UsdGeom.Cube.Define(stage,path)
    geom.CreateSizeAttr(1.0)
    geom.AddTranslateOp().Set(Gf.Vec3d(*pos))
    geom.AddScaleOp().Set(Gf.Vec3f(*scale))
    geom.CreateDisplayColorAttr([Gf.Vec3f(*color)])
    UsdPhysics.CollisionAPI.Apply(geom.GetPrim())
    if dynamic:
        UsdPhysics.RigidBodyAPI.Apply(geom.GetPrim())
        UsdPhysics.MassAPI.Apply(geom.GetPrim()).CreateMassAttr(0.01)
    return geom

box('/World/Table',(0,0,-0.025),(1.2,1.0,0.05),(0.65,0.63,0.58))
box('/World/Brick',(0.02,-0.23,0.015),(0.032,0.016,0.018),(0.8,0.03,0.02),True)
# Low open trays approximate the destinations; no grasp or task success is implied.
for name,x,color in [('Orange',-.20,(1.,.35,.02)),('Purple',.0,(.5,.05,.8)),('Yellow',.20,(1.,.8,.02))]:
    y=.20
    box('/World/'+name+'Tray/Base',(x,y,.004),(.12,.12,.008),color)
    for i,(dx,dy,sx,sy) in enumerate([(-.06,0,.008,.12),(.06,0,.008,.12),(0,-.06,.12,.008),(0,.06,.12,.008)]):
        box('/World/'+name+f'Tray/Wall{i}',(x+dx,y+dy,.022),(sx,sy,.04),color)

def camera(path, eye, target):
    cam=UsdGeom.Camera.Define(stage,path)
    cam.CreateClippingRangeAttr(Gf.Vec2f(.005,20.))
    cam.CreateFocalLengthAttr(18.)
    cam.CreateHorizontalApertureAttr(24.)
    cam.CreateVerticalApertureAttr(18.)
    transform=Gf.Matrix4d().SetLookAt(Gf.Vec3d(*eye),Gf.Vec3d(*target),Gf.Vec3d(0,0,1)).GetInverse()
    cam.AddTransformOp().Set(transform)
    return str(cam.GetPath())

camera_paths={
    'top':camera('/World/TopCamera',(.0,-.12,.8),(0,0,0)),
    'inspect':camera('/World/InspectCamera',(.48,-.3,.24),(0,-.14,.16)),
    'overview':camera('/World/OverviewCamera',(.7,-.8,.65),(0,0,.13)),
}
gripper_paths=[str(p.GetPath()) for p in stage.Traverse() if p.GetName()=='gripper_link']
print('GRIPPER_LINKS',gripper_paths)
assert gripper_paths, 'No gripper link found'
camera_paths['wrist']=camera(gripper_paths[0]+'/WristCamera',(.03,-.04,.02),(.03,-.16,-.08))
annotators={}
render_products={}
for name,path in camera_paths.items():
    product=rep.create.render_product(path,(640,480))
    ann=rep.AnnotatorRegistry.get_annotator('rgb')
    ann.attach([product])
    render_products[name]=product
    annotators[name]=ann

SimulationManager.setup_simulation(dt=1/60,device='cpu')
from isaacsim.core.experimental.prims import Articulation
robot=Articulation('/World/SO101')
timeline=omni.timeline.get_timeline_interface()
timeline.play()
await app_utils.update_app_async(steps=120)
indices=robot.get_dof_indices(JOINTS).numpy().reshape(-1).tolist()
lower_limits,upper_limits=robot.get_dof_limits()
limits=np.stack([lower_limits.numpy()[0],upper_limits.numpy()[0]],axis=-1)
print('DOF_INDICES',indices,'LIMITS',limits.tolist())
initial=np.zeros((1,len(limits)),dtype=np.float32)
initial[0,indices[-1]]=0.8
robot.set_dof_positions(initial)
robot.set_dof_position_targets(initial)
await app_utils.update_app_async(steps=120)
timeline.pause()
stage.GetRootLayer().Export(str(OUT/'scene.usda'))
from PIL import Image
for name,ann in annotators.items():
    pixels=np.asarray(ann.get_data())[:,:,:3]
    assert pixels.size and np.std(pixels)>1, (name,pixels.shape)
    Image.fromarray(pixels).save(OUT/f'initial-{name}.png')
(OUT/'scene-ready.json').write_text(json.dumps({'usd':str(usd_path),'camera_paths':camera_paths,'joint_indices':indices,'limits':limits.tolist(),'calibration':'URDF new midpoint convention; real-unit alignment unverified'},indent=2))
print('SCENE_READY',flush=True)
