"""Physical AI Studio policy worker; file IPC isolates CUDA dependencies from Kit."""
import hashlib
import inspect
import json
import os
import sys
import time
from pathlib import Path
import numpy as np
import torch
from transformers.initialization import no_init_weights
from physicalai.policies import Pi05
from physicalai.data import Observation

ROOT=Path('/work')
OUT=ROOT/'outputs'
IPC=ROOT/'ipc'
EXPECTED_SIZE=12180881108
EXPECTED_SHA='27218d38cf69df81a448794a8484dc0e7cc6f2b15151ddde5ac5625ec9381d85'
checkpoint=ROOT/'model/best.ckpt'
print('RUNTIME',torch.__version__,torch.cuda.get_device_name(0),flush=True)
deadline=time.monotonic()+1800
while not checkpoint.exists() or checkpoint.stat().st_size<EXPECTED_SIZE:
    if time.monotonic()>deadline: raise TimeoutError('Checkpoint download incomplete')
    time.sleep(2)
assert checkpoint.stat().st_size==EXPECTED_SIZE
with checkpoint.open('rb') as f: digest=hashlib.file_digest(f,'sha256').hexdigest()
assert digest==EXPECTED_SHA, digest
print('CHECKPOINT_SHA256_OK',digest,flush=True)
torch.manual_seed(42)
# mmap leaves optimizer tensors on disk. assign avoids copying the model state
# into a second randomly initialized parameter allocation.
saved=torch.load(checkpoint,map_location='cpu',mmap=True,weights_only=False)
hparams={k:v for k,v in saved['hyper_parameters'].items() if k in inspect.signature(Pi05.__init__).parameters}
hparams.update(pretrained_name_or_path=None,compile_model=False,gradient_checkpointing=False)
print('CONSTRUCTING_POLICY',sorted(hparams),flush=True)
with no_init_weights():
    policy=Pi05(**hparams)
policy.on_load_checkpoint(saved)
policy.load_state_dict(saved['state_dict'],strict=True,assign=True)
del saved
import gc
gc.collect()
print('STATE_LOADED',flush=True)
policy=policy.eval().to('cuda')
policy.requires_grad_(False)
# Exercise the exact tokenizer before reporting readiness.
tokenizer=policy._preprocessor.tokenizer
stats=policy.hparams['dataset_stats']
metadata={'torch':torch.__version__,'gpu':torch.cuda.get_device_name(0),
    'checkpoint_sha256':digest,'model_revision':'34e0f9d11f9b91fa07743493b11b76ac80961675',
    'dataset_stats':stats,'tokenizer':str(type(tokenizer)),
    'camera_order':['wrist','top','inspect'],'joint_order':['shoulder_pan','shoulder_lift','elbow_flex','wrist_flex','wrist_roll','gripper']}
(OUT/'policy-ready.json').write_text(json.dumps(metadata,indent=2,default=str))
print('POLICY_READY',flush=True)
for step in range(10):
    obsfile=IPC/f'obs-{step:03d}.npz'
    deadline=time.monotonic()+900
    while not obsfile.exists():
        if time.monotonic()>deadline: raise TimeoutError(str(obsfile))
        time.sleep(.05)
    data=np.load(obsfile,allow_pickle=False)
    obs=Observation(
        state=torch.from_numpy(data['state'].copy()).unsqueeze(0),
        images={name:torch.from_numpy(data[name].copy()).permute(2,0,1).unsqueeze(0).float()/255 for name in ['wrist','top','inspect']},
        task=[str(data['instruction'].item())],
    )
    torch.cuda.synchronize()
    started=time.perf_counter()
    with torch.inference_mode(),torch.autocast('cuda',dtype=torch.bfloat16):
        actions=policy.predict_action_chunk(obs)
    torch.cuda.synchronize()
    latency=time.perf_counter()-started
    actions=actions.float().cpu().numpy()[0]
    assert actions.shape==(50,6),actions.shape
    assert np.isfinite(actions).all()
    tmp=IPC/f'actions-{step:03d}.tmp.npz'
    np.savez(tmp,actions=actions,latency_s=latency)
    os.replace(tmp,IPC/f'actions-{step:03d}.npz')
    print(json.dumps({'chunk':step,'latency_s':latency,'min':actions.min(axis=0).tolist(),'max':actions.max(axis=0).tolist()}),flush=True)
print('WORKER_COMPLETE',flush=True)
